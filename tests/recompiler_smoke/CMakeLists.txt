    string(REPLACE "-" "_" target_suffix "${name}")

  # A mutant copies the conversion header and the two it includes so no unmutated copy is reachable.
  function(add_native_convert_mutant name old_text new_text)
    set(header_file native_kernel_convert.h)
    file(READ "${SUYU_SOURCE}/src/core/native_memory/${header_file}" header_text)
    string(FIND "${header_text}" "${old_text}" first_at)
    string(FIND "${header_text}" "${old_text}" last_at REVERSE)
    if(first_at EQUAL -1 OR NOT first_at EQUAL last_at)
      message(FATAL_ERROR "native convert mutant ${name}: text must occur exactly once in ${header_file}: ${old_text}")
    endif()
    string(REPLACE "${old_text}" "${new_text}" mutated_text "${header_text}")
    set(mutant_dir "${CMAKE_CURRENT_BINARY_DIR}/native_convert_mutants/${name}")
    foreach(copied_header native_address_space.h native_process_memory.h)
      file(READ "${SUYU_SOURCE}/src/core/native_memory/${copied_header}" copied_text)
      file(WRITE "${mutant_dir}/core/native_memory/${copied_header}" "${copied_text}")
    endforeach()
    file(WRITE "${mutant_dir}/core/native_memory/${header_file}" "${mutated_text}")
    string(REPLACE "-" "_" target_suffix "${name}")
    add_executable(smoke_native_${target_suffix} native_kernel_convert_unit.cpp)
    target_compile_features(smoke_native_${target_suffix} PRIVATE cxx_std_20)
    target_include_directories(smoke_native_${target_suffix} PRIVATE "${mutant_dir}" "${SUYU_SOURCE}/src")
  endfunction()
